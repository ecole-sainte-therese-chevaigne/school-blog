<iframe
  id="school-github-blog"
  data-src="https://ecole-sainte-therese-chevaigne.github.io/school-blog/"
  title="La vie de l’école Sainte-Thérèse"
  loading="lazy"
  scrolling="no"
  style="width:100%; height:1100px; border:0; display:block; overflow:hidden;"
  allow="fullscreen">
</iframe>
<script>
(function () {
  var frame = document.getElementById('school-github-blog');
  if (!frame) return;

  var blogUrl = new URL(frame.getAttribute('data-src'));
  var category = new URLSearchParams(window.location.search).get('category');
  if (category) blogUrl.searchParams.set('category', category);

  var blogOrigin = blogUrl.origin;

  window.addEventListener('message', function (event) {
    if (event.origin !== blogOrigin || event.source !== frame.contentWindow) return;
    if (!event.data || event.data.type !== 'school-blog-height') return;

    var height = Number(event.data.height);
    if (!Number.isFinite(height) || height < 300) return;

    frame.style.height = Math.ceil(height) + 'px';
  });

  frame.src = blogUrl.toString();
})();
</script>
